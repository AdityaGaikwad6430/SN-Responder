#This is installation of Tomcat


And these are the steps and commands 

-> 1) sudo apt update
   2) sudo apt install -y openjdk-17-jdk

#Java successfully installed

#Create and add a Tomcat user

  1) sudo useradd -m -U -d /opt/tomcat -s /bin/false tomcat


# Installing Tomcat

  1) sudo wget https://dlcdn.apache.org/tomcat/tomcat-11/v11.0.5/bin/apache-tomcat-11.0.5.tar.gz -P /tmp

  2) sudo tar -xvf /tmp/apache-tomcat-11.0.5.tar.gz -C /opt/tomcat

  3) sudo chown -R tomcat:tomcat /opt/tomcat

  4) cd /etc/systemd/system

  5) sudo nano tomcat.service
    # Add these lines as it is in the edited file
    #[Unit] , [Service] , [Install] Should be green
      

[Unit]

Description=Tomcat Server
After=network.target

[Service]

Type=forking
User=tomcat
Group=tomcat
Environment="JAVA_HO ME=/usr/lib/jvm/java-17-openjdk-amd64"
WorkingDirectory=/opt/tomcat/apache-tomcat-11.0.5
ExecStart=/opt/tomcat/apache-tomcat-11.0.5/bin/startup.sh

[Install]

WantedBy=multi-user.target



  6) sudo systemctl daemon-reload

  7) sudo systemctl start tomcat                (This starts Tomcat)
  8) sudo systemctl enable tomcat               (This starts Tomcat on startup automatically)    


9) sudo ufw allow 8080                 (This allows port number 8080 for global)
